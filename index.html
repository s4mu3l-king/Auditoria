async function processarEEnviarPlanilha() { const input = $("input-excel"); const divStatus = $("status-importacao"); if (!input.files || input.files.length === 0) { return alert("Por favor, selecione um arquivo de planilha."); } const file = input.files[0]; divStatus.innerText = "Lendo e validando dados da planilha..."; const reader = new FileReader(); reader.onload = async function (e) { try { const data = new Uint8Array(e.target.result); const workbook = XLSX.read(data, { type: 'array' }); const firstSheetName = workbook.SheetNames[0]; const worksheet = workbook.Sheets[firstSheetName]; const rows = XLSX.utils.sheet_to_json(worksheet, { header: 1 }); const registrosParaInserir = []; for (let i = 1; i < rows.length; i++) { const row = rows[i]; if (!row || row.length === 0) continue; const colA = row[0]; // Filial const colB = String(row[1] || '').trim(); // Coluna B const colC = String(row[2] || '').trim(); // Coluna C const colE = row[4]; // Depósito const colG = row[6]; // Cultura / Cultivar const colI = String(row[8] || '').trim(); // Descrição Tratamento const colL = row[11]; // Lote const colN = row[13]; // Categoria (Cat) const colO = row[14]; // Penera (Pen) const colX = row[23]; // Peso Etiqueta const colZ = row[25]; // Unidade (UM) const colAE = parseFloat(row[30] || 0); // Qtd. Sistema (Coluna AE) // Filtro Principal: AE > 0 if (colAE <= 0 || isNaN(colAE)) continue; // De/Para de Local e UF let localFinal = ""; let ufFinal = ""; if (colB === "0101" && colC === "10") { localFinal = "UBS I"; ufFinal = "GO"; } else if (colB === "102" && colC === "10") { localFinal = "LPCO"; ufFinal = "GO"; } else if (colB.includes("BJ") || colC.includes("BJ")) { localFinal = "BJ"; ufFinal = "MG"; } else if (colB.includes("AGROSEMEAR") || colC.includes("AGROSEMEAR")) { localFinal = "AGROSEMEAR"; ufFinal = "MT"; } else { localFinal = colB || colC; ufFinal = "GO"; } // Sigla de Tratamento (última palavra da Coluna I) const partesTratamento = colI.split(/\s+/); const ultimaSigla = partesTratamento[partesTratamento.length - 1].toUpperCase(); const tratamentoFinal = SIGLAS_TRATAMENTO_VALIDAS.includes(ultimaSigla) ? ultimaSigla : "ST"; registrosParaInserir.push({ local: localFinal, uf: ufFinal, filial: colA, // Corrigido para 'filial' deposito: colE, cultura: colG, lote: colL, cultivar: colG, Tratamento: tratamentoFinal, cat: colN, pen: colO, um: colZ, qtd_sistema: colAE, peso_etiqueta: colX }); } if (registrosParaInserir.length === 0) { divStatus.innerText = "Nenhum registro válido encontrado com Coluna AE > 0."; return; } divStatus.innerText = `Salvando ${registrosParaInserir.length} registros no Supabase...`; const { error } = await db.from('base_dados').upsert(registrosParaInserir); if (error) throw error; divStatus.innerText = "✅ Importação concluída com sucesso!"; await dashboard(); setTimeout(() => { fecharModalImportacao(); }, 1500); } catch (err) { console.error(err); divStatus.innerText = "❌ Erro ao processar planilha: " + err.message; } }; reader.readAsArrayBuffer(file); }
